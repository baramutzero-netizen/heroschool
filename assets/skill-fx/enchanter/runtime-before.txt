/* 그 순간의 투명도 · 변형 — t: 효과가 보이는 동안(0 ~ 1) · beat: 한 박자(0 ~ 1 비율) · 진영 B(오른쪽 보스)가 그림 그대로, A 는 좌우 반전 */
function bossUltFxAt(B, t, beat, side){
  t = clamp(t, 0, 1);
  let o = 1, tf = "";
  if(B.anim === "barrage"){
    let dx = 0;
    if(t < .06){ o = t/.06; dx = .1*(1 - t/.06); }
    else if(t < .8){ const ph = Math.floor((t - .06)/Math.max(.01, beat)) % 2; o = ph ? .75 : 1; dx = ph ? .01 : -.02; }
    else { const k = (t - .8)/.2; o = 1 - k; dx = -.02 - .04*k; }
    tf = `translateX(${(dx*100).toFixed(2)}%)`;
  } else if(B.anim === "slash"){
    let dx = 0, dy = 0, s = 1;
    if(t < .1){ const k = t/.1; o = k; dx = .07*(1 - k); dy = -.07*(1 - k); s = .8 + .2*k; }
    else if(t >= .55){ const k = (t - .55)/.45; o = 1 - k; dx = -.04*k; dy = .04*k; }
    tf = `translate(${(dx*100).toFixed(2)}%, ${(dy*100).toFixed(2)}%) scale(${s.toFixed(3)})`;
  } else {
    let s;
    if(t < .1){ const k = t/.1; o = k; s = .45 + .49*k; }
    else if(t < .6){ s = .94 + .06*(t - .1)/.5; }
    else { const k = (t - .6)/.4; o = 1 - k; s = 1 + .12*k; }
    tf = `scale(${s.toFixed(3)})`;
  }
  return {o: clamp(o, 0, 1), tf: (side === "A" ? "scaleX(-1) " : "") + tf};
}

function stylishFormation(actor, allies, enemies, side, skillId){   // 1024 — 숫자는 STYLISH_TUNE (STLAB · 편집 툴 tools/stylish_lab.html)
  const T=stT(), positions=[], mirror=x=>side==="B"?100-x:x;
  const lane=(list,[left,right])=>list.forEach((uid,i)=>positions.push({uid,
    x:mirror(list.length===1?(left+right)/2:left+(right-left)*i/(list.length-1)),
    y:T.y, scale:T.scale}));
  const sk=stSkillLane(skillId, enemies.length);
  if(sk){lane([actor,...allies],sk.ally);lane(enemies,sk.enemy);}
  else if(enemies.length){lane([actor,...allies],T.ally);lane(enemies,T.enemy);}
  else {lane([actor,...allies],T.solo);}
  return positions;
}
